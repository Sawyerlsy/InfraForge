### 开机自动执行脚本
开机启动时执行控制脚本script_runner.sh，该脚本会按数字前缀顺序执行/opt/scripts下的脚本，并记录执行日志。
### 配置步骤

#### 1、创建脚本
```shell
# 在/usr/local/bin目录下创建控制脚本
vim /usr/local/bin/script_runner.sh
```

```shell
#!/bin/bash
# 描述：服务器启动时按数字前缀顺序执行/opt/scripts下的脚本
# 作者：Sawyerlsy | 日期：2025-03-06
# 主脚本位置:/usr/local/bin/script_runner.sh
# 子脚本位置:/opt/scripts
LOG_FILE="/var/log/script_runner.log"
MAX_LOG_SIZE=10485760  # 10MB日志自动清理
SCRIPT_DIR="/opt/scripts"
FAIL_COUNT=0           # 失败计数器
# 清理超过10MB的旧日志
if [ -f "$LOG_FILE" ] && [ $(stat -c%s "$LOG_FILE") -gt $MAX_LOG_SIZE ]; then
    echo "[$(date '+%F %T')] 清理历史日志" > "$LOG_FILE"
fi
# 检查目录权限
if [ ! -d "$SCRIPT_DIR" ] || [ ! -r "$SCRIPT_DIR" ] || [ ! -x "$SCRIPT_DIR" ]; then
    echo "[$(date '+%F %T')] [ERROR] 目录不可访问: $SCRIPT_DIR" | tee -a "$LOG_FILE"
    exit 1
fi
# 按数字前缀排序执行
while IFS= read -r script; do
    # 校验脚本权限
    if [ ! -x "$script" ]; then
        echo "[$(date '+%F %T')] [ERROR] 无执行权限: $script" | tee -a "$LOG_FILE"
        ((FAIL_COUNT++))
        continue
    fi
    # 记录执行开始
    echo "[$(date '+%F %T')] [START] $script" | tee -a "$LOG_FILE"
    start_time=$(date +%s)
    # 带超时执行（默认2分钟）
    timeout 120 bash "$script" >> "$LOG_FILE" 2>&1
    retval=$?
    # 记录结果
    end_time=$(date +%s)
    duration=$((end_time - start_time))
    
    if [ $retval -eq 0 ]; then
        status="成功"
    elif [ $retval -eq 124 ]; then
        status="超时"
        ((FAIL_COUNT++))
    else
        status="失败(code:$retval)"
        ((FAIL_COUNT++))
    fi
    echo "[$(date '+%F %T')] [END] $script | 状态: $status | 耗时: ${duration}秒" | tee -a "$LOG_FILE"
done < <(find "$SCRIPT_DIR" -maxdepth 1 -name '[0-9][0-9]_*.sh' | sort -n)  # 严格数字排序[3](@ref)
# 最终状态码
exit $((FAIL_COUNT > 0 ? 1 : 0))
```

#### 2、创建服务文件
```shell
# 在/etc/systemd/systems目录下创建服务文件
vim /etc/systemd/system/script_runner.service
```

```shell
[Unit]
Description=Execute all scripts 
After=network.target syslog.target  # 增加日志服务依赖
Requires=network-online.target     # 确保网络就绪

[Service]
User=root
Group=root
Type=oneshot
ExecStart=/bin/bash /usr/local/bin/script_runner.sh
TimeoutStartSec=300                # 全局超时5分钟
RemainAfterExit=yes
StandardOutput=journal
StandardError=journal              # 错误日志统一到journal

[Install]
WantedBy=multi-user.target
```

#### 3、设置权限和启动服务
```shell
sudo chmod +x /usr/local/bin/script_runner.sh
sudo chmod 744 /etc/systemd/system/script_runner.service
sudo systemctl daemon-reload
sudo systemctl enable script_runner.service
```

#### 4、手动测试服务
```shell
sudo systemctl start script_runner.service
journalctl -u script_runner.service -f
```

#### 5、查看详细日志
```shell
tail -f /var/log/script_runner.log
```
